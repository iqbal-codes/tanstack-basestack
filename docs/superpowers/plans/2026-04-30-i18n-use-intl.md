# i18n with use-intl Implementation Plan

> **For agentic workers:** Use inline execution with checkpoints.

**Goal:** Add EN (default, no prefix) + ID (/id/ prefix) translations using use-intl + TanStack Router rewrite.

**Architecture:** use-intl library with namespaced message objects. Router `rewrite` strips/adds `/id/` prefix. `{-$locale}` layout validates locale. `createIsomorphicFn` detects locale server/client. IntlProvider in root shell. Admin routes bypass locale prefix (read from cookie).

**Tech Stack:** use-intl (~2kb), TanStack Router rewrite, createIsomorphicFn, React 19

- [ ] **Step 1: Create src/lib/i18n.ts** — shared config

```ts
import { defaultLocale, supportedLocales, type Locale } from '#/messages'

export { defaultLocale, supportedLocales }
export type { Locale }

export const LOCALE_COOKIE = 'locale'

export function isIgnoredPath(pathname: string): boolean {
  return /^\/admin(?:\/|$)/.test(pathname) || /^\/api(?:\/|$)/.test(pathname)
}

export function extractLocaleFromPath(pathname: string): Locale | null {
  const match = /^\/([a-z]{2})(?:\/|$)/.exec(pathname)
  const locale = match?.[1]
  if (!locale) return null
  if (locale === defaultLocale) return null
  return supportedLocales.includes(locale as Locale)
    ? (locale as Locale)
    : null
}

export function stripLocale(pathname: string): string {
  const locale = extractLocaleFromPath(pathname)
  if (!locale) return pathname
  return pathname.replace(`/${locale}`, '') || '/'
}

export function parseLocaleCookie(cookieHeader: string | null): string | null {
  if (!cookieHeader) return null
  const match = cookieHeader.match(new RegExp(`${LOCALE_COOKIE}=([^;]+)`))
  return match?.[1] ?? null
}
```

- [ ] **Step 2: Create src/lib/i18n.client.ts** — rewrite fns + locale detection

```ts
import { createIsomorphicFn } from '@tanstack/react-start'
import { getRequest } from '@tanstack/react-start/server'
import {
  defaultLocale,
  extractLocaleFromPath,
  isIgnoredPath,
  LOCALE_COOKIE,
  parseLocaleCookie,
  stripLocale,
  type Locale,
} from './i18n'

export function deLocalizeUrl(url: URL): URL {
  if (isIgnoredPath(url.pathname)) return url
  const newUrl = new URL(url)
  const stripped = stripLocale(url.pathname)
  newUrl.pathname = stripped
  return newUrl
}

export function localizeUrl(url: URL, currentLocale: string): URL {
  if (isIgnoredPath(url.pathname)) return url
  if (currentLocale === defaultLocale) return url
  const newUrl = new URL(url)
  newUrl.pathname = `/${currentLocale}${url.pathname === '/' ? '' : url.pathname}`
  return newUrl
}

export const getCurrentLocale = createIsomorphicFn()
  .server(() => {
    const request = getRequest()
    const url = new URL(request.url)
    if (isIgnoredPath(url.pathname)) {
      return parseLocaleCookie(request.headers.get('cookie')) ?? defaultLocale
    }
    return extractLocaleFromPath(url.pathname) ?? defaultLocale
  })
  .client(() => {
    const path = window.location.pathname
    if (isIgnoredPath(path)) {
      return parseLocaleCookie(document.cookie) ?? defaultLocale
    }
    return extractLocaleFromPath(path) ?? defaultLocale
  })
```

```tsx
import { createRouter as createTanStackRouter } from '@tanstack/react-router'
import { setupRouterSsrQueryIntegration } from '@tanstack/react-router-ssr-query'
import NotFound from '#/components/not-found'
import { getContext } from './integrations/tanstack-query/root-provider'
import { routeTree } from './routeTree.gen'
import { deLocalizeUrl, localizeUrl } from '#/lib/i18n.client'
import { getCurrentLocale } from '#/lib/i18n.client'
import { defaultLocale } from '#/lib/i18n'

export function getRouter() {
  const context = getContext()

  const router = createTanStackRouter({
    routeTree,
    context,
    scrollRestoration: true,
    defaultPreload: 'intent',
    defaultPreloadStaleTime: 0,
    defaultNotFoundComponent: NotFound,
    rewrite: {
      input: ({ url }) => deLocalizeUrl(url),
      output: ({ url }) => localizeUrl(url, getCurrentLocale() ?? defaultLocale),
    },
  })

  setupRouterSsrQueryIntegration({ router, queryClient: context.queryClient })

  return router
}

declare module '@tanstack/react-router' {
  interface Register {
    router: ReturnType<typeof getRouter>
  }
}
```

### Task 4: Wire IntlProvider in root

- [ ] **Step 1: Update __root.tsx** — Add IntlProvider + dynamic lang

```tsx
import { IntlProvider } from 'use-intl'
import { TanStackDevtools } from '@tanstack/react-devtools'
import type { QueryClient } from '@tanstack/react-query'
import {
  createRootRouteWithContext,
  HeadContent,
  Outlet,
  Scripts,
} from '@tanstack/react-router'
import { TanStackRouterDevtoolsPanel } from '@tanstack/react-router-devtools'
import { TooltipProvider } from '#/components/ui/tooltip'
import TanStackQueryDevtools from '../integrations/tanstack-query/devtools'
import appCss from '../styles.css?url'
import { getCurrentLocale } from '#/lib/i18n.client'
import { messages } from '#/messages'

interface MyRouterContext {
  queryClient: QueryClient
}

export const Route = createRootRouteWithContext<MyRouterContext>()({
  head: () => ({
    meta: [
      { charSet: 'utf-8' },
      { name: 'viewport', content: 'width=device-width, initial-scale=1' },
      { title: 'Admin Console' },
    ],
    links: [{ rel: 'stylesheet', href: appCss }],
  }),
  shellComponent: RootDocument,
  component: Outlet,
})

function RootDocument({ children }: { children: React.ReactNode }) {
  const locale = getCurrentLocale()

  return (
    <html lang={locale} suppressHydrationWarning>
      <head>
        <HeadContent />
      </head>
      <body
        className="font-sans antialiased wrap-anywhere"
        suppressHydrationWarning
      >
        <IntlProvider locale={locale} messages={messages[locale ?? 'en']}>
          <TooltipProvider>{children}</TooltipProvider>
        </IntlProvider>
        <TanStackDevtools
          config={{ position: 'bottom-right' }}
          plugins={[
            {
              name: 'Tanstack Router',
              render: <TanStackRouterDevtoolsPanel />,
            },
            TanStackQueryDevtools,
          ]}
        />
        <Scripts />
      </body>
    </html>
  )
}
```
